Créer une table “restaurants” contenant les champs suivants :
Id,
Nom,
Type_cuisine,
Localisation,
Prix_moyen.

CREATE TABLE restaurants(
    Id int AUTO_INCREMENT PRIMARY KEY,
    Nom varchar(100),
    Type_cuisine varchar(100),
    Localisation varchar(100),
    Prix_moyen int);

Ecrire la requête SQL qui affiche uniquement les noms des restaurants localisés à Paris.

SELECT Nom FROM restaurants
WHERE Localisation = "Paris";

Ecrire la requête SQL qui affiche la liste des types de cuisines, toutes villes confondues (sans les doublons).

SELECT DISTINCT Type_cuisine FROM restaurants;

Ecrire la requête SQL qui affiche la liste des restaurants dont le prix moyen est inférieur à 50€.

SELECT * FROM restaurants
WHERE Prix_moyen < 50;

Ecrire la requête SQL qui affiche le nombre de restaurants à Paris.

SELECT COUNT(*) FROM restaurants
WHERE Localisation = "Paris";

Ecrire la requête SQL qui affiche les noms des restaurants qui sont présents plusieurs fois

SELECT Nom FROM restaurants
GROUP BY Nom
HAVING COUNT(*) >1;

Ecrire la requête SQL qui affiche les restaurants par ordre alphabétique de nom.

SELECT Nom FROM restaurants
ORDER BY Nom ASC;

Ecrire la requête SQL qui affiche le nombre de restaurants par ville.

SELECT Localisation, COUNT(*) FROM restaurants
GROUP by Localisation;
